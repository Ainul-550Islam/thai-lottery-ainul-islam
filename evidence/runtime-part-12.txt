    public function index(Request $request, ?string $reference = null): View\n
    {\n
        $panel = $this->panelFor($request);\n
        $this->authorizePanel($request, $panel);\n
\n
        $payload = match ($panel) {\n
            'draws' => ['records' => $this->drawProjection()],\n
            'bets' => ['records' => $this->betProjection()],\n
            'wallets' => ['records' => $this->walletProjection()],\n
            'ledger' => ['records' => $this->transactionProjection()],\n
            'audits' => ['records' => $this->auditProjection($request)],\n
            'kyc' => ['records' => $this->kycProjection($request->user())],\n
            'claims' => ['records' => $this->claimProjection($reference)],\n
            'freezes' => ['records' => $this->freezeProjection($reference)],\n
            default => [],\n
        };\n
\n
        return view('admin.dashboard', [\n
            'panel' => $panel,\n
            'kpis' => $panel === 'dashboard' ? $this->calculateExecutiveKpis() : [],\n
            'transactions' => $panel === 'dashboard' ? $this->fetchFinancialFeed() : [],\n
            'records' => $payload['records'] ?? [],\n
            'state' => in_array($panel, [\n
                'reconciliation', 'compliance', 'risk', 'payments', 'withdrawals',\n
                'settlements', 'wallet_operations', 'payment_methods', 'withdrawal_methods',\n
                'payment_events', 'payment_exceptions', 'draw_lifecycle', 'publication',\n
                'imports', 'sources', 'lotteries', 'lottery_rules', 'fees', 'account_grades',\n
                'account_verification', 'responsible_gaming', 'self_exclusion', 'users',\n
                'user_detail', 'user_finance', 'bet_detail', 'ticket_detail',\n
                'ticket_verification', 'claim_review', 'commissions', 'health', 'queues',\n
                'scheduler', 'runtime', 'api_status', 'webhooks', 'security', 'release',\n
                'cutover',\n
            ], true)\n
                ? 'NOT_CONFIGURED'\n
                : null,\n
        ]);\n
    }\n

    public function analyticsApi(Request $request): JsonResponse\n
    {\n
        $this->authorizePanel($request, 'dashboard');\n
        [$from, $to] = $this->boundedPeriod($request);\n
\n
        return response()->json([\n
            'status' => 'success',\n
            'period' => [\n
                'from' => $from->toDateString(),\n
                'to' => $to->toDateString(),\n
            ],\n
            'kpis' => $this->calculateExecutiveKpis($from, $to),\n
        ]);\n
    }\n
\n
    public function reconciliationFeedApi(Request $request): JsonResponse\n
    {\n
        $this->authorizePanel($request, 'reconciliation');\n
\n
        // A bank/provider balance feed is not configured in this application.\n
        // Returning an explicit state is safer than presenting an internal\n
        // journal as if it were an external settlement balance.\n
        return response()->json([\n
            'status' => 'NOT_CONFIGURED',\n
            'data' => [],\n
        ], 200);\n
    }\n

    public function runReconciliation(Request $request): JsonResponse\n
    {\n
        $this->authorizePanel($request, 'reconciliation');\n
        [$from, $to] = $this->boundedPeriod($request);\n
        $operator = $request->user();\n
\n
        $report = $this->reconciliationService->reconcile(\n
            from: $from->startOfDay(),\n
            to: $to->endOfDay(),\n
            initiatedBy: $operator !== null ? (string) $operator->getAuthIdentifier() : null,\n
        );\n
\n
        return response()->json([\n
            'status' => 'success',\n
            'report' => $report->toArray(),\n
        ]);\n
    }\n
\n
    public function downloadKyc(Request $request, string $documentToken): Response\n
    {\n
        $this->authorizePanel($request, 'kyc');\n
        $viewer = $request->user();\n
\n
        if ($viewer === null) {\n
            abort(401);\n
        }\n
\n
        $document = $this->accountVerification->documentForReviewer($viewer, $documentToken);\n
        if (! $document instanceof KycDocument) {\n
            abort(404);\n
        }\n
\n
        $file = $this->documents->readForAuthorized($document, $viewer, true);\n
\n
        return response($file['contents'], 200, [\n
            'Content-Type' => $file['mime'],\n
            'Content-Disposition' => 'attachment; filename="'.$file['name'].'"',\n
            'X-Content-Type-Options' => 'nosniff',\n
            'Cache-Control' => 'private, no-store',\n
        ]);\n
    }\n
\n
    public function reviewKyc(Request $request, string $documentToken): Response\n
    {\n
        $this->authorizePanel($request, 'kyc');\n
        $decision = str_contains((string) $request->route()?->getName(), '.reject') ? 'reject' : 'approve';\n
        $validated = $request->validate([\n
            'reason' => ['nullable', 'string', 'max:1000'],\n
        ]);\n
        $viewer = $request->user();\n
\n
        if ($viewer === null) {\n
            abort(401);\n
        }\n
\n
        $document = $this->accountVerification->documentForReviewer($viewer, $documentToken);\n
        if (! $document instanceof KycDocument) {\n
            abort(404);\n
        }\n
\n
        $approved = $decision === 'approve';\n
        if (! in_array($decision, ['approve', 'reject'], true)) {\n
            abort(404);\n
        }\n
\n
        $this->accountVerification->review(\n
            document: $document,\n
            reviewer: $viewer,\n
            approved: $approved,\n
            reason: isset($validated['reason']) ? (string) $validated['reason'] : null,\n
        );\n
\n
        return redirect()->back()->with('status', trans('admin.review_recorded'));\n
    }\n

    /**\n
     * @return array<string, string|int>\n
     */\n
    private function calculateExecutiveKpis(?Carbon $from = null, ?Carbon $to = null): array\n
    {\n
        $currency = Currency::tryFrom(strtoupper((string) config('finance.currency.default', Currency::THB->value))) ?? Currency::THB;\n
        $bets = Bet::query()\n
            ->where('currency', $currency->value)\n
            ->whereNotIn('status', ['cancelled', 'refunded'])\n
            ->when($from !== null, static fn ($query) => $query->where('placed_at', '>=', $from))\n
            ->when($to !== null, static fn ($query) => $query->where('placed_at', '<=', $to));\n
\n
        $hasBets = (clone $bets)->exists();\n
        $totalWagered = $hasBets ? $this->formatMoney((string) ((clone $bets)->sum('stake_amount')), $currency) : 'NO_DATA';\n
\n
        $withdrawals = Withdrawal::query()\n
            ->where('currency', $currency->value)\n
            ->whereIn('status', ['completed', 'COMPLETED'])\n
            ->when($from !== null, static fn ($query) => $query->where('completed_at', '>=', $from))\n
            ->when($to !== null, static fn ($query) => $query->where('completed_at', '<=', $to));\n
\n
        $activeBets = Bet::query()\n
            ->where('currency', $currency->value)\n
            ->whereIn('status', ['active', 'pending'])\n
            ->count();\n
        $pendingWithdrawals = Withdrawal::query()\n
            ->where('currency', $currency->value)\n
            ->whereIn('status', ['pending', 'under_review', 'kyc_required'])\n
            ->count();\n
\n
        return [\n
            'totalWagered' => $totalWagered,\n
            'totalWageredTrend' => 'UNAVAILABLE',\n
            'houseGrossProfit' => 'UNAVAILABLE',\n
            'houseGrossProfitTrend' => 'UNAVAILABLE',\n
            'activeInPlayBets' => $activeBets,\n
            'activeInPlayBetsState' => $activeBets === 0 ? 'NO_DATA' : 'AVAILABLE',\n
            'pendingWithdrawals' => $pendingWithdrawals,\n
            'pendingWithdrawalsState' => $pendingWithdrawals === 0 ? 'NO_DATA' : 'AVAILABLE',\n
            'completedWithdrawalsTotal' => $withdrawals->exists()\n
                ? $this->formatMoney((string) $withdrawals->sum('amount'), $currency)\n
                : 'NO_DATA',\n
        ];\n
    }\n

    /**\n
     * @return list<array<string, string|null>>\n
     */\n
    private function fetchFinancialFeed(): array\n
    {\n
        return FinancialTransaction::query()\n
            ->latest('id')\n
            ->limit(100)\n
            ->get(['reference_number', 'type', 'status', 'currency', 'amount', 'created_at'])\n
            ->map(static fn (FinancialTransaction $transaction): array => [\n
                'reference' => (string) $transaction->reference_number,\n
                'type' => $transaction->type instanceof \BackedEnum ? (string) $transaction->type->value : (string) $transaction->type,\n
                'status' => $transaction->status instanceof \BackedEnum ? (string) $transaction->status->value : (string) $transaction->status,\n
                'currency' => $transaction->currency instanceof \BackedEnum ? (string) $transaction->currency->value : (string) $transaction->currency,\n
                'amount' => (string) $transaction->amount,\n
                'created_at' => $transaction->created_at?->toIso8601String(),\n
            ])\n
            ->all();\n
    }\n

    /**\n
     * @return list<array<string, mixed>>\n
     */\n
    private function kycProjection(?User $reviewer): array\n
    {\n
        if (! $reviewer instanceof User) {\n
            return [];\n
        }\n
\n
        return KycDocument::query()->latest('id')->limit(100)->get([\n
            'id', 'user_id', 'document_type', 'status', 'created_at',\n
        ])->map(function (KycDocument $document): array {\n
            $token = $this->accountVerification->reviewerDocumentToken($document, $reviewer);\n
\n
            return [\n
                'reference' => 'UNAVAILABLE',\n
                'type' => $document->document_type instanceof \BackedEnum ? (string) $document->document_type->value : (string) $document->document_type,\n
                'status' => $document->status instanceof \BackedEnum ? (string) $document->status->value : (string) $document->status,\n
                'currency' => null,\n
                'amount' => null,\n
                'created_at' => $document->created_at?->toIso8601String(),\n
                'download_url' => route('admin.kyc.download', ['documentToken' => $token]),\n
                'approve_url' => route('admin.kyc.approve', ['documentToken' => $token]),\n
                'reject_url' => route('admin.kyc.reject', ['documentToken' => $token]),\n
            ];\n
        })->all();\n
    }\n

    /**\n
     * @return array{0: Carbon, 1: Carbon}\n
     */\n
    private function boundedPeriod(Request $request): array\n
    {\n
        $validated = $request->validate([\n
            'from' => ['nullable', 'date_format:Y-m-d'],\n
            'to' => ['nullable', 'date_format:Y-m-d'],\n
        ]);\n
        $to = Carbon::createFromFormat('Y-m-d', (string) ($validated['to'] ?? now()->toDateString()))->endOfDay();\n
        $from = Carbon::createFromFormat('Y-m-d', (string) ($validated['from'] ?? $to->copy()->subDays(30)->toDateString()))->startOfDay();\n
\n
        if ($from->greaterThan($to) || $from->diffInDays($to) > 31) {\n
            abort(422, trans('admin.analytics_range_invalid'));\n
        }\n
\n
        return [$from, $to];\n
    }\n

    private function authorizePanel(Request $request, string $panel): void\n
    {\n
        $user = $request->user();\n
        $permission = match ($panel) {\n
            'draws' => AdminAccess::VIEW_DRAWS,\n
            'risk', 'compliance' => AdminAccess::VIEW_RISK_ALERTS,\n
            'payments', 'withdrawals' => AdminAccess::MANAGE_PAYOUTS,\n
            'bets' => AdminAccess::VIEW_TRANSACTION_HISTORY,\n
            'audits' => AdminAccess::VIEW_AUDIT_LOGS,\n
            'wallets' => AdminAccess::MANAGE_WALLET,\n
            'ledger' => AdminAccess::VIEW_FINANCIAL_REPORTS,\n
            'reconciliation' => AdminAccess::RECONCILE_LEDGER,\n
            'kyc' => AdminAccess::MANAGE_USERS,\n
            'claims', 'claim_review' => AdminAccess::MANAGE_GLO_PRIZE_CLAIMS,\n
            'freezes' => AdminAccess::REVIEW_GLO_FREEZES,\n
            'settlements' => AdminAccess::PROCESS_SETTLEMENTS,\n
            'wallet_operations' => AdminAccess::MANAGE_WALLET,\n
            'payment_methods', 'withdrawal_methods', 'payment_events', 'payment_exceptions' => AdminAccess::MANAGE_PAYOUTS,\n
            'draw_lifecycle' => AdminAccess::MANAGE_DRAWS,\n
            'publication', 'imports', 'sources' => AdminAccess::VIEW_RESULTS,\n
            'lotteries', 'lottery_rules', 'fees', 'account_grades' => AdminAccess::MANAGE_SYSTEM_SETTINGS,\n
            'account_verification', 'responsible_gaming', 'self_exclusion', 'users', 'user_detail', 'user_finance', 'bet_detail', 'ticket_detail', 'ticket_verification' => AdminAccess::MANAGE_USERS,\n
            'commissions' => AdminAccess::VIEW_COMMISSIONS,\n
            'queues', 'scheduler', 'runtime', 'api_status', 'webhooks', 'security', 'release', 'cutover', 'health' => AdminAccess::VIEW_AUDIT_LOGS,\n
            default => AdminAccess::VIEW_DASHBOARD,\n
        };\n
\n
        if (! AdminAccess::canAccessPanel($user) || ! AdminAccess::allows($user, $permission)) {\n
            abort(403, trans('admin.access_denied'));\n
        }\n
    }\n

    private function formatMoney(string $amount, Currency $currency): string\n
